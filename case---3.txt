name = input("Enter patient name: ")

consultation = float(input("Consultation charges: "))
diagnostic = float(input("Diagnostic charges: "))
treatment = float(input("Treatment charges: "))

total = consultation + diagnostic + treatment

if total >= 10000:
    discount = total * 0.10
else:
    discount = 0

final_bill = total - discount

print("\nHOSPITAL BILL")
print("Patient Name:", name)
print("Consultation:", consultation)
print("Diagnostic:", diagnostic)
print("Treatment:", treatment)
print("Total Amount:", total)
print("Discount:", discount)
print("Final Bill:", final_bill)

paid = float(input("Enter amount paid: "))

if paid < 0 or paid > final_bill:
    print("Invalid payment amount")
else:
    balance = final_bill - paid

    if balance == 0:
        print("Patient Classification: Fully Paid")
    elif paid > 0:
        print("Patient Classification: Partially Paid")
    else:
        print("Patient Classification: Unpaid")

    print("Balance Amount:", balance)
